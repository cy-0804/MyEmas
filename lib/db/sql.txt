CREATE TABLE public.users (
  user_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  fullname text,
  gender text,
  date_of_birth date,
  address text,
  email text UNIQUE,
  icno text,
  password_hash text,
  phone_num text,
  language text,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  role_id USER-DEFINED,
  CONSTRAINT users_pkey PRIMARY KEY (user_id)
);
CREATE TABLE public.elderly (
  user_id uuid NOT NULL,
  chronic_condition text,
  mobility_status text,
  blood_type text,
  allergies text,
  CONSTRAINT elderly_pkey PRIMARY KEY (user_id),
  CONSTRAINT elderly_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(user_id)
);
CREATE TABLE public.caregiver (
  user_id uuid NOT NULL,
  organization text,
  availability text,
  CONSTRAINT caregiver_pkey PRIMARY KEY (user_id),
  CONSTRAINT caregiver_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(user_id)
);
CREATE TABLE public.care_link (
  link_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  elderly_id uuid,
  caregiver_id uuid,
  relationship text,
  emergency_contact_primary boolean DEFAULT false,
  CONSTRAINT care_link_pkey PRIMARY KEY (link_id),
  CONSTRAINT care_link_elderly_id_fkey FOREIGN KEY (elderly_id) REFERENCES public.elderly(user_id),
  CONSTRAINT care_link_caregiver_id_fkey FOREIGN KEY (caregiver_id) REFERENCES public.caregiver(user_id)
);
CREATE TABLE public.health_record (
  record_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  elderly_id uuid NOT NULL,
  blood_pressure text,
  heart_rate integer,
  glucose_level double precision,
  temperature double precision,
  recorded_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT health_record_pkey PRIMARY KEY (record_id),
  CONSTRAINT health_record_elderly_id_fkey FOREIGN KEY (elderly_id) REFERENCES public.elderly(user_id)
);
CREATE TABLE public.health_risk_assessment (
  risk_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  record_id uuid,
  risk_level USER-DEFINED,
  recommendation text,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT health_risk_assessment_pkey PRIMARY KEY (risk_id),
  CONSTRAINT health_risk_assessment_record_id_fkey FOREIGN KEY (record_id) REFERENCES public.health_record(record_id)
);
CREATE TABLE public.schedule (
  schedule_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  elderly_id uuid,
  title text,
  schedule_date_time timestamp without time zone,
  location text,
  notes text,
  CONSTRAINT schedule_pkey PRIMARY KEY (schedule_id),
  CONSTRAINT schedule_elderly_id_fkey FOREIGN KEY (elderly_id) REFERENCES public.elderly(user_id)
);
CREATE TABLE public.reminder (
  reminder_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  type text,
  message text,
  reminder_time timestamp without time zone,
  CONSTRAINT reminder_pkey PRIMARY KEY (reminder_id)
);
CREATE TABLE public.medications (
  medication_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  medication_name text,
  dosage text,
  when_to_take text,
  instruction text,
  medication_stock integer,
  medical_notes text,
  schedule_id uuid,
  expiration_date date,
  priority text,
  photo text,
  CONSTRAINT medications_pkey PRIMARY KEY (medication_id),
  CONSTRAINT medications_schedule_id_fkey FOREIGN KEY (schedule_id) REFERENCES public.schedule(schedule_id)
);
CREATE TABLE public.medication_logs (
  log_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  medication_id uuid,
  status USER-DEFINED,
  logged_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT medication_logs_pkey PRIMARY KEY (log_id),
  CONSTRAINT medication_logs_medication_id_fkey FOREIGN KEY (medication_id) REFERENCES public.medications(medication_id)
);
CREATE TABLE public.emergency_logs (
  alert_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  link_id uuid,
  timestamp timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  status text,
  location text,
  caregiver_id uuid,
  CONSTRAINT emergency_logs_pkey PRIMARY KEY (alert_id),
  CONSTRAINT emergency_logs_link_id_fkey FOREIGN KEY (link_id) REFERENCES public.care_link(link_id),
  CONSTRAINT emergency_logs_caregiver_id_fkey FOREIGN KEY (caregiver_id) REFERENCES public.caregiver(user_id)
);
CREATE TABLE public.allergy (
  allergy_id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL,
  type text,
  symptoms text,
  CONSTRAINT allergy_pkey PRIMARY KEY (allergy_id)
);
CREATE TABLE public.allergy_list (
  allergy_id uuid NOT NULL,
  elderly_id uuid NOT NULL,
  recommendation text,
  CONSTRAINT allergy_list_pkey PRIMARY KEY (allergy_id, elderly_id),
  CONSTRAINT allergy_list_allergy_id_fkey FOREIGN KEY (allergy_id) REFERENCES public.allergy(allergy_id),
  CONSTRAINT allergy_list_elderly_id_fkey FOREIGN KEY (elderly_id) REFERENCES public.elderly(user_id)
);
